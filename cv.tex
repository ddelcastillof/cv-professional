% Vancouver/ICMJE output corrections:
% - drop language ("en.") and urldate ("[Accessed on: ...]") from entries
% - drop url: the DOI is already printed, and vancouver.bbx would otherwise
%   append "Available from: <url>" to every entry
% - drop month/day: Paperpile writes legacy values like month = 15~2, which
%   biber flags and vancouver.bbx would render as a stray "2026 Apr"
% - comma (not "and") before the final author
% - ragged-right bibliography: stops justification stretching DOI/URL breaks
\AtEveryBibitem{\clearlist{language}\clearfield{langid}\clearfield{urldate}\clearfield{urlday}\clearfield{urlmonth}\clearfield{urlyear}\clearfield{urldateera}%
  \clearfield{url}\clearfield{month}\clearfield{day}}
\renewcommand*{\finalnamedelim}{\multinamedelim}
\appto{\bibsetup}{\raggedright}

% vancouver.bbx's volume+number+eid macro prints \printfield{issue}, but
% Paperpile writes the issue into `number`, so the issue silently vanished
% (e.g. "An Fac med 2021; 81" instead of "81(4)"). Print `number` when
% `issue` is absent, formatted the same way vancouver.bbx formats `issue`.
% Journal titles italic. vancouver.bbx:72 prints them upright (strict ICMJE);
% the old pandoc template emphasised them too, though `ulem` turned that
% emphasis into an underline. Delete this line for upright ICMJE journals.
\DeclareFieldFormat*{journaltitle}{\mkbibemph{#1}}

\DeclareFieldFormat*[article]{number}{(#1)}
\renewbibmacro*{volume+number+eid}{%
  \printfield{volume}%
  \iffieldundef{issue}{\printfield{number}}{\printfield{issue}}%
  \printfield{eid}%
}

% Own name as literal text (addendum notes etc.). The \mkbibnamefamily hook
% below only fires on parsed name lists, never on plain fields, so anywhere the
% name is typed as free text it must be bolded through this macro instead.
% Must be defined *before* \DeclareSourcemap: biblatex expands the fieldvalue
% when it writes the sourcemap to build/cv.bcf, not at bibliography time.
\newcommand*{\ownnamebf}{\textbf{Del Castillo Fernandez, D.}}

% Field patches Paperpile will never carry.
%
% maxnames=6 truncates long author lists to "et al", which hides the own name
% on large collaborations. Vidyasagaran2024-wx has 61 authors with Del Castillo
% at position 26, so the addendum states the contribution explicitly. Add one
% \map block per affected cite key — only needed when the own name sits beyond
% position 6.
\DeclareSourcemap{%
  \maps[datatype=bibtex]{%
    \map{%
      \step[fieldsource=entrykey, match={Vidyasagaran2024-wx}, final]
      \step[fieldset=addendum,
            fieldvalue={Among authors: \ownnamebf}]%
    }%
    % Conference presentations, NLM form:
    %   Author. Title. Presented at: Event; Year Mon; City, Country.
    % Three things stop vancouver.bbx producing that on its own:
    %   - its @misc driver reads neither eventtitle nor venue, so the event
    %     vanishes; addendum is the field it does print (see Vidyasagaran above)
    %   - \AtEveryBibitem clears month/day (the Paperpile "month = 15~2" fix),
    %     so date={2026-09-11} renders as a bare "2026" — the month and day
    %     have to be written into addendum by hand to survive. Keep them in
    %     sync if Paperpile revises the entry's date.
    %   - a bare year would then print twice, so date is nulled here
    % "and" not "&": a literal ampersand is an alignment tab, and \& would have
    % to survive biblatex serialising the fieldvalue into build/cv.bcf.
    % The title's "Rapid-Fire Presentation:" prefix is Paperpile's; stripped so
    % the format marker sits where NLM puts it, next to the event.
    \map{%
      \step[fieldsource=entrykey, match={Del-Castillo2026-jq}, final]
      \step[fieldsource=title,
            match=\regexp{Rapid-Fire\s+Presentation:\s*}, replace={}]
      \step[fieldset=addendum,
            fieldvalue={Rapid-fire presentation at: 70th Society for Social
                        Medicine and Population Health Conference -- European
                        Congress of Epidemiology; 2026 Sep 11; London, UK
                        (forthcoming)}]%
      \step[fieldset=date, null]%
    }%
  }%
}

% Bold own name (replaces filters/bold-author.lua).
%
% Paperpile spells the name inconsistently across entries — "Del Castillo,
% Darwin" in four, "Del Castillo-Fernández, Darwin" in one — so the test keys
% on the family-name prefix, which every spelling shares.
%
% \namepartfamily is a macro whose tokens are joined by delimiter *macros*, and
% biblatex picks which one by token count, verified with \meaning:
%   Del\bibnamedelima Castillo
%   Del\bibnamedelimb Castillo\bibnamedelima Fernandez
%   Del\bibnamedelimb Río\bibnamedelima Castillo
% So no fixed-string test matches the raw macro. Flatten every delimiter to a
% plain space, \edef the result, then test the *ordered* prefix.
%
% Ordering is what makes this safe against new entries. An unordered
% "contains Del AND contains Castillo" test also bolds coauthors who merely
% share substrings — "Delgado Castillo, Diego", "Castillo Delgado, Daniel",
% "Del Río Castillo, Diana" all match it, and bib/references.bib already
% carries a "Delgado, Carlos A" one position ahead of the own name in
% Lapidaire2023-pr. The prefix test rejects all three and still accepts
% "Del Castillo", "Del Castillo Fernandez" and "Del Castillo-Fernández".
%
% Residual limitation: a coauthor named "Del Castillo, D…" (say Diana) would
% still match. Tighten \ownnamegiveninitial to {Darwin} if that ever happens —
% at the cost of missing entries that abbreviate the given name to "D".
\usepackage{xstring}
\newif\ifownname
\newcommand*{\ownnamefamilyprefix}{Del Castillo}
\newcommand*{\ownnamegiveninitial}{D}
\makeatletter
\newcommand*{\checkownname}{%
  \global\ownnamefalse
  \ifdefvoid{\namepartgiven}{}{%
    \ifdefvoid{\namepartfamily}{}{%
      \begingroup
        \def\bibnamedelima{ }\def\bibnamedelimb{ }\def\bibnamedelimc{ }%
        \def\bibnamedelimd{ }\def\bibnamedelimi{ }\def\bibhyphendelim{-}%
        \protected@edef\ownnameflatfamily{\namepartfamily}%
        \protected@edef\ownnameflatgiven{\namepartgiven}%
        \expandarg
        \IfBeginWith{\ownnameflatfamily}{\ownnamefamilyprefix}
          {\IfBeginWith{\ownnameflatgiven}{\ownnamegiveninitial}
            {\global\ownnametrue}{}}%
          {}%
      \endgroup
    }%
  }%
}
\makeatother
\renewcommand*{\mkbibnamefamily}[1]{\checkownname\ifownname\textbf{#1}\else#1\fi}
\renewcommand*{\mkbibnamegiven}[1]{\checkownname\ifownname\textbf{#1}\else#1\fi}
